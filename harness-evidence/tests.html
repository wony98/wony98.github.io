25def run_helper(*args: str) -> subprocess.CompletedProcess[str]:
26    return subprocess.run(
27        [sys.executable, str(HELPER), *args],
28        capture_output=True,
29        check=False,
30        text=True,
31        encoding="utf-8",
32    )

35def helper_version() -> str:
36    result = run_helper("--version")
37    if result.returncode != 0:
38        raise AssertionError(result.stderr or "helper --version failed")
39    return result.stdout.strip()

42def assert_pointer_version(pointer: Path, expected: str) -> bool:
43    if not pointer.exists():
44        return False
45    content = pointer.read_text(encoding="utf-8")
46    declared = POINTER_VERSION_RE.findall(content)
47    if not declared:
48        raise AssertionError("Claude pointer has no declared helper version")
49    if set(declared) != {expected}:
50        raise AssertionError(
51            f"Claude pointer versions {sorted(set(declared))} do not match helper {expected}"
52        )
53    return True

56def initialize(thread: Path, mode: str | None = None) -> subprocess.CompletedProcess[str]:
57    args = ["--thread", str(thread), "init", "--title", "Regression"]
58    if mode is not None:
59        args.extend(("--mode", mode))
60    return run_helper(*args)

63def append_message(
64    thread: Path,
65    expected_last: int,
66    body: str = "body",
67    expected_mode: str | None = None,
68    reply: str = "Y",
69    in_reply_to: str | None = None,
70    sender: str = "CODEX",
71    receiver: str = "CLAUDE/OWNER",
72    status: str = "OPEN",
73    artifact: str | None = None,
74    message_type: str = "REPORT",
75) -> subprocess.CompletedProcess[str]:
76    args = [
77        "--thread",
78        str(thread),
79        "append",
80        "--expect-last",
81        str(expected_last),
82        "--tag",
83        "REGRESSION",
84        "--type",
85        message_type,
86        "--sender",
87        sender,
88        "--receiver",
89        receiver,
90        "--title",
91        "message",
92        "--body",
93        body,
94        "--reply",
95        reply,
96        "--status",
97        status,
98    ]
99    if expected_mode is not None:
100        args.extend(("--expect-mode", expected_mode))
101    if in_reply_to is not None:
102        args.extend(("--in-reply-to", in_reply_to))
103    if artifact is not None:
104        args.extend(("--artifact", artifact))
105    return run_helper(*args)

108def marker_block(raw: bytes, start: bytes, end: bytes) -> bytes:
109    start_index = raw.index(start)
110    end_index = raw.index(end, start_index)
111    end_line = raw.find(b"\n", end_index)
112    end_after = len(raw) if end_line == -1 else end_line + 1
113    return raw[start_index:end_after]

116def message_region(raw: bytes) -> bytes:
117    index = raw.find(b"### [")
118    return b"" if index == -1 else raw[index:]

323        with tempfile.TemporaryDirectory() as temp_dir:

326            def malformed(name: str, transform) -> subprocess.CompletedProcess[str]:
327                thread = base / f"{name}.md"

341                ),
342                "invalid_enum": lambda raw: raw.replace(b"- \xeb\xaa\xa8\xeb\x93\x9c: GIT", b"- \xeb\xaa\xa8\xeb\x93\x9c: git", 1),
343                "duplicate_field": lambda raw: raw.replace(
344                    b"- \xeb\xaa\xa8\xeb\x93\x9c: GIT\n",

346                    1,
347                ),
348                "mode_after_pin": lambda raw: marker_block(raw, PIN_START, PIN_END)
349                + b"\n"
350                + marker_block(raw, MODE_START, MODE_END)
351                + message_region(raw),
352            }
