86def parse_headers(text: str) -> list[tuple[int, int]]:
87    found: list[tuple[int, int]] = []
88    for line_no, line in enumerate(text.splitlines(), start=1):
89        match = HEADER_RE.match(line.strip())
90        if match:
91            found.append((int(match.group("num")), line_no))
92    return found

108def first_message_offset(raw: bytes) -> int:
109    """Return the first numbered-message byte offset, or EOF when none exists."""
110    offset = 0
111    for raw_line in raw.splitlines(keepends=True):
112        line = decode_utf8(raw_line).strip()
113        if HEADER_RE.match(line):
114            return offset
115        offset += len(raw_line)
116    return len(raw)

119def marker_region_in_prefix(
120    prefix: bytes,
121    start_text: str,
122    end_text: str,
123    label: str,
124) -> MarkerRegion | None:
125    start_marker = start_text.encode("ascii")
126    end_marker = end_text.encode("ascii")
127    start_count = prefix.count(start_marker)
128    end_count = prefix.count(end_marker)
129    if start_count == 0 and end_count == 0:
130        return None
131    if start_count != 1 or end_count != 1:
132        raise ValueError(
133            f"{label} requires exactly one start marker and one end marker "
134            f"in the pre-message region (start={start_count}, end={end_count})"
135        )
136    start_index = prefix.index(start_marker)
137    end_index = prefix.index(end_marker)
138    if end_index <= start_index:
139        raise ValueError(f"{label} markers are out of order")
140    if start_index and prefix[start_index - 1 : start_index] != b"\n":
141        raise ValueError(f"{label} start marker must begin on its own line")
142    if end_index and prefix[end_index - 1 : end_index] != b"\n":
143        raise ValueError(f"{label} end marker must begin on its own line")
144    start_line_end = prefix.find(b"\n", start_index)
145    if start_line_end == -1 or start_line_end >= end_index:
146        raise ValueError(f"{label} start marker has no line ending")
147    end_line_end = prefix.find(b"\n", end_index)
148    end_after = len(prefix) if end_line_end == -1 else end_line_end + 1
149    body_start = start_line_end + 1
150    body = prefix[body_start:end_index]
151    return MarkerRegion(start_index, body_start, end_index, end_after, body)

154def thread_layout(raw: bytes) -> ThreadLayout:
155    first_message = first_message_offset(raw)
156    prefix = raw[:first_message]
157    mode = marker_region_in_prefix(prefix, MODE_START, MODE_END, "mode")
158    pin_region = marker_region_in_prefix(prefix, PIN_START, PIN_END, "pin")
159    if mode is not None and pin_region is None:
160        raise ValueError("mode block requires a separate pinned block after it")
161    if mode is not None and pin_region is not None and mode.end_after > pin_region.start:
162        raise ValueError("mode block must appear before the pinned block without nesting")
163    return ThreadLayout(first_message=first_message, mode=mode, pin=pin_region)

166def parse_mode_body(body: bytes) -> tuple[str, str, str]:
167    text = decode_utf8(body).replace("\r\n", "\n").replace("\r", "\n")
168    lines = [line for line in text.split("\n") if line]
169    prefixes = ("- 모드: ", "- 설정: ", "- 사유: ")
170    values: dict[str, str] = {}
171    for prefix in prefixes:
172        matching = [line for line in lines if line.startswith(prefix)]
173        if len(matching) != 1:
174            raise ValueError(f"mode field {prefix.strip()!r} occurs {len(matching)} times")
175        values[prefix] = matching[0][len(prefix) :]
176    if len(lines) != len(prefixes):
177        raise ValueError("mode block contains unexpected or duplicate fields")
178    mode_value = values["- 모드: "]
179    if mode_value not in MODES:
180        raise ValueError(f"invalid mode value {mode_value!r}; expected GIT or LOCAL")
181    setting = values["- 설정: "]
182    setter = setting.split(" · ", 1)[0].strip()
183    reason = values["- 사유: "].strip()
184    if not setter or not reason:
185        raise ValueError("mode setter and reason must not be empty")
186    return mode_value, setter, reason

189def mode_info(raw: bytes, *, reject_message_only_markers: bool = False) -> ModeInfo | None:
190    layout = thread_layout(raw)
191    if layout.mode is None:
192        if reject_message_only_markers:
193            message_region = raw[layout.first_message :]
194            if MODE_START.encode("ascii") in message_region or MODE_END.encode("ascii") in message_region:
195                raise ValueError("mode markers exist only in the message region")
196        return None
197    value, setter, reason = parse_mode_body(layout.mode.body)
198    return ModeInfo(
199        value=value,
200        setter=setter,
201        reason=reason,
202        body_hash=hashlib.sha256(layout.mode.body).hexdigest(),
203    )

206def validate_raw(raw: bytes, expected_last: int | None = None) -> tuple[bool, list[str], int]:
207    text = decode_utf8(raw)
208    ok, problems, last = validate_text(text, expected_last=expected_last)
209    try:
210        layout = thread_layout(raw)
211        if layout.mode is not None:

213    except ValueError as error:
214        problems.append(str(error))
215    return not problems and ok, problems, last

218def newline_counts(text: str) -> tuple[int, int, int]:
219    crlf = text.count("\r\n")
220    without_crlf = text.replace("\r\n", "")
221    lf_only = without_crlf.count("\n")
222    stray_cr = without_crlf.count("\r")
223    return crlf, lf_only, stray_cr

226def newline_label(text: str) -> str:
227    crlf, lf_only, stray_cr = newline_counts(text)
228    if crlf and lf_only:
229        return "MIXED"
230    if crlf:
231        return "CRLF"
232    if lf_only:
233        return "LF"
234    return "NONE"

244def metadata_field_lines(
245    lines: list[str], start: int, end: int, prefix: str
246) -> list[int]:
247    """Return matching rows only from the header metadata block, never the body."""
248    matches: list[int] = []

262def parse_structured_reply_targets(value: str) -> tuple[int, ...]:
263    stripped = value.strip()
264    if not re.fullmatch(r"#\d{4}(?:\s*,\s*#\d{4})*", stripped):
265        raise ValueError(
266            "reply targets must use #NNNN comma-separated form, for example #0001,#0002"
267        )
268    numbers = tuple(int(match.group("num")) for match in REPLY_NUMBER_RE.finditer(stripped))
269    if len(numbers) != len(set(numbers)):
270        raise ValueError("reply targets must not contain duplicates")
271    return numbers

274def parse_cli_reply_targets(value: str | None) -> tuple[int, ...]:
275    if value is None:
276        return ()
277    stripped = value.strip()
278    if not re.fullmatch(r"\d+(?:\s*,\s*\d+)*", stripped):
279        raise ValueError("--in-reply-to must be a comma-separated list of message numbers")
280    numbers = tuple(int(part.strip()) for part in stripped.split(","))
281    if len(numbers) != len(set(numbers)):
282        raise ValueError("--in-reply-to must not contain duplicate message numbers")
283    return numbers

286def legacy_reply_targets(lines: list[str], body_start: int, end: int) -> tuple[int, ...]:
287    numbers: list[int] = []
288    for index in range(body_start, end):
289        stripped = lines[index].rstrip("\r\n")
290        if not LEGACY_REPLY_LINE_RE.fullmatch(stripped):
291            continue
292        numbers.extend(int(match.group("num")) for match in REPLY_NUMBER_RE.finditer(stripped))
293    return tuple(dict.fromkeys(numbers))

296def message_infos(text: str) -> list[MessageInfo]:
297    lines = text.splitlines(keepends=True)
298    found = parse_headers(text)
299    existing_numbers = {number for number, _ in found}
300    messages: list[MessageInfo] = []
301    for index, (number, line_no) in enumerate(found):
302        start = line_no - 1
303        end = found[index + 1][1] - 1 if index + 1 < len(found) else len(lines)
304        header = lines[start].rstrip("\r\n")
305        detail = HEADER_DETAIL_RE.match(header)
306        if detail is None:
307            sender, receiver, title = "?", "?", header
308        else:
309            sender = detail.group("sender").strip()
310            receiver = detail.group("receiver").strip()
311            title = detail.group("title").strip()
312
313        reply_rows = metadata_field_lines(lines, start, end, "- 답필요: ")
314        reply_required = len(reply_rows) == 1 and (
315            lines[reply_rows[0]].rstrip("\r\n").removeprefix("- 답필요: ") == "Y"
316        )
317
318        state_rows = metadata_state_lines(lines, start, end)
319        status = (
320            lines[state_rows[0]].rstrip("\r\n").removeprefix("- 상태: ")
321            if len(state_rows) == 1
322            else ""
323        )
324
325        target_rows = metadata_field_lines(lines, start, end, "- 회신대상: ")
326        if target_rows:
327            value = lines[target_rows[0]].rstrip("\r\n").removeprefix("- 회신대상: ")
328            targets = parse_structured_reply_targets(value)
329        else:
330            body_marker = next(
331                (
332                    row
333                    for row in range(start + 1, end)
334                    if lines[row].rstrip("\r\n") == "- 본문:"
335                ),
336                None,
337            )
338            body_start = body_marker + 1 if body_marker is not None else end
339            targets = legacy_reply_targets(lines, body_start, end)
340
341        valid_targets = tuple(
342            target for target in targets if target in existing_numbers and target < number
343        )
344        messages.append(
345            MessageInfo(
346                number=number,
347                line_no=line_no,
348                sender=sender,
349                receiver=receiver,
350                title=title,
351                status=status,
352                reply_required=reply_required,
353                reply_targets=valid_targets,
354            )
355        )
356    return messages

359def message_block(text: str, number: int) -> str:
360    """Return one complete numbered message without reading unrelated history."""
361    lines = text.splitlines(keepends=True)
362    found = parse_headers(text)
363    for index, (candidate, line_no) in enumerate(found):
364        if candidate != number:
365            continue
366        start = line_no - 1
367        end = found[index + 1][1] - 1 if index + 1 < len(found) else len(lines)
368        return "".join(lines[start:end]).rstrip("\r\n")
369    raise ValueError(f"message not found: #{number:04d}")

372def message_body(text: str, number: int) -> str:
373    block = message_block(text, number)
374    marker = "- 본문:"
375    lines = block.splitlines()
376    try:
377        index = lines.index(marker)
378    except ValueError:

380    return "\n".join(lines[index + 1 :]).strip()

383def receiver_matches(receiver: str, agent: str | None) -> bool:
384    if not agent:
385        return True
386    tokens = {
387        token
388        for token in re.split(r"[^A-Za-z0-9_-]+", receiver.upper())
389        if token
390    }
391    wanted = agent.upper()
392    return wanted in tokens or "ALL" in tokens

396    if limit < 1 or len(value) <= limit:

402def read_only_validation(raw: bytes, command: str) -> tuple[str, int] | None:
403    text = decode_utf8(raw)
404    ok, problems, last = validate_raw(raw)
405    structural = [
406        problem for problem in problems if not problem.startswith("mixed line endings:")
407    ]
408    if structural:
409        for problem in structural:
410            print(problem, file=sys.stderr)
411        return None
412    if not ok:
413        print(
414            f"[thread] warning: mixed newlines detected; read-only {command} is allowed, "
415            "mutation commands will refuse",
416            file=sys.stderr,
417        )
418    return text, last

421def intake(path: Path, args: argparse.Namespace) -> int:
422    """Emit one bounded mailbox snapshot for the common read/process path."""
423    raw = read_bytes(path)
424    validated = read_only_validation(raw, "intake")
425    if validated is None:
426        return 1
427    text, last = validated
428    messages = message_infos(text)
429    awaiting_all = awaiting_reply_messages(text)
430    relevant_pending = [
431        message
432        for message in messages
433        if message.status in PENDING_STATUSES
434        and receiver_matches(message.receiver, args.agent)
435    ]
436    relevant_awaiting = [
437        message
438        for message in awaiting_all
439        if receiver_matches(message.receiver, args.agent)
440    ]
441
442    latest_by_status: dict[str, MessageInfo] = {}
443    for message in relevant_pending:
444        latest_by_status[message.status] = message
445
446    incremental_summary: str | None = None
447    if args.known_last is not None:
448        if args.known_last < 0:
449            print("[thread] intake refused: --known-last must be 0 or greater", file=sys.stderr)
450            return 2
451        if args.known_last > last:
452            print(
453                f"[thread] intake refused: --known-last #{args.known_last:04d} "
454                f"is greater than actual last #{last:04d}",
455                file=sys.stderr,
456            )
457            return 2
458        if args.limit < 1:
459            print(
460                "[thread] intake refused: --limit must be at least 1 with --known-last",
461                file=sys.stderr,
462            )
463            return 2
464        incremental = [
465            message
466            for message in messages
467            if message.number > args.known_last
468            and receiver_matches(message.receiver, args.agent)
469        ]
470        shown = incremental[: args.limit]
471        remaining = len(incremental) - len(shown)
472        next_after = shown[-1].number if shown else args.known_last
473        next_range = (
474            f"#{incremental[len(shown)].number:04d}-#{incremental[-1].number:04d}"
475            if remaining
476            else "NONE"
477        )
478        incremental_summary = (
479            f"[thread] incremental agent={args.agent} after=#{args.known_last:04d} "
480            f"matched={len(incremental)} shown={len(shown)} remaining={remaining} "
481            f"next_after=#{next_after:04d} next_range={next_range}"
482        )
483        selected_numbers = {message.number for message in shown}
484    else:
485        selected_numbers = {
486            message.number for message in relevant_pending[-max(0, args.limit) :]
487        }
488        selected_numbers.update(
489            message.number for message in latest_by_status.values()
490        )
491        if args.recent:
492            selected_numbers.update(
493                message.number for message in messages[-max(0, args.recent) :]
494            )
495    selected = [
496        message for message in messages if message.number in selected_numbers
497    ]
498
499    info = mode_info(raw)
500    mode_label = info.value if info is not None else "UNDECLARED"
501    layout = thread_layout(raw)
502    pin_digest = (
503        hashlib.sha256(layout.pin.body).hexdigest()
504        if layout.pin is not None
505        else "NONE"
506    )
507    if args.known_pinned_hash is None:
508        pinned_changed = "UNKNOWN"
509    else:
510        pinned_changed = (
511            "NO"
512            if args.known_pinned_hash.strip().lower() == pin_digest.lower()
513            else "YES"
514        )
515    latest_labels = " ".join(
516        f"latest_{status}=#{latest_by_status[status].number:04d}"
517        if status in latest_by_status
518        else f"latest_{status}=NONE"
519        for status in PENDING_STATUSES
520    )
521    print(
522        f"[thread] intake mode={mode_label} last=#{last:04d} messages={len(messages)} "
523        f"pinned_hash={pin_digest} pinned_changed={pinned_changed} "
524        f"pending_for={len(relevant_pending)} awaiting_for={len(relevant_awaiting)} "
525        f"awaiting_total={len(awaiting_all)}"
526    )
527    print(f"[thread] {latest_labels}")
528    if incremental_summary is not None:
529        print(incremental_summary)
530
531    if not args.no_pinned:
532        if layout.pin is None:
533            print("[PINNED] unavailable")
534        elif pinned_changed == "NO":
535            print("[PINNED] unchanged; body omitted")
536        else:
537            pin_text = decode_utf8(layout.pin.body).strip()
538            print(f"[PINNED sha256={pin_digest}]")
539            print(clipped(pin_text, args.pinned_chars) or "(empty)")
540
541    if not selected:
542        print("[MESSAGES] none selected")
543        return 0
544    for message in selected:
545        flags = []
546        if message in relevant_awaiting:
547            flags.append("AWAITING")
548        if message.status in PENDING_STATUSES:
549            flags.append(message.status)
550        flag_text = ",".join(dict.fromkeys(flags)) or message.status or "UNKNOWN"
551        print(
552            f"[MESSAGE #{message.number:04d} {flag_text}] "
553            f"{message.sender} -> {message.receiver} | {message.title}"
554        )
555        print(clipped(message_body(text, message.number), args.body_chars) or "(empty body)")
556    return 0

559def linked_message_numbers(
560    messages: list[MessageInfo], number: int, depth: int
561) -> list[int]:
562    by_number = {message.number: message for message in messages}
563    if number not in by_number:
564        raise ValueError(f"message not found: #{number:04d}")
565    ordered: list[int] = []
566    seen: set[int] = set()
567
568    def visit(current: int, remaining: int) -> None:
569        if current in seen:
570            return
571        seen.add(current)
572        message = by_number[current]
573        if remaining > 0:
574            for target in message.reply_targets:
575                visit(target, remaining - 1)
576        ordered.append(current)
577
578    visit(number, max(0, depth))
579    return ordered

622def validate_text(text: str, expected_last: int | None = None) -> tuple[bool, list[str], int]:
623    problems: list[str] = []
624    crlf, lf_only, stray_cr = newline_counts(text)
625    if crlf and lf_only:
626        problems.append(f"mixed line endings: crlf={crlf}, lf_only={lf_only}")
627    # Standalone CR bytes can occur inside preserved historical message bodies.
628    # Report their count, but reject only an actual CRLF/LF newline mixture.
629
630    found = parse_headers(text)
631    if not found:
632        if expected_last not in (None, 0):
633            problems.append(f"expected last #{expected_last:04d}, got #{0:04d}")
634        return not problems, problems, 0
635
636    seen: dict[int, list[int]] = {}
637    for number, line_no in found:
638        seen.setdefault(number, []).append(line_no)
639    for number, line_numbers in sorted(seen.items()):
640        if len(line_numbers) > 1:
641            problems.append(f"duplicate message #{number:04d}: lines {line_numbers}")
642
643    numbers = [number for number, _ in found]
644    for previous, current in zip(numbers, numbers[1:]):
645        if current <= previous:
646            problems.append(f"message order inversion: #{previous:04d} then #{current:04d}")
647    last = numbers[-1]
648    if last != max(numbers):
649        problems.append(f"last message #{last:04d} is not maximum #{max(numbers):04d}")
650    if expected_last is not None and last != expected_last:
651        problems.append(f"expected last #{expected_last:04d}, got #{last:04d}")
652
653    lines = text.splitlines(keepends=True)
654    for index, (number, line_no) in enumerate(found):
655        start = line_no - 1
656        end = found[index + 1][1] - 1 if index + 1 < len(found) else len(lines)
657        state_rows = metadata_state_lines(lines, start, end)
658        if len(state_rows) != 1:
659            problems.append(f"message #{number:04d} has {len(state_rows)} metadata status rows")
660            continue
661        state_value = lines[state_rows[0]].rstrip("\r\n").removeprefix("- 상태: ")
662        if state_value not in STATUSES:
663            problems.append(f"message #{number:04d} has invalid status {state_value!r}")
664
665        reply_rows = metadata_field_lines(lines, start, end, "- 답필요: ")
666        if len(reply_rows) != 1:
667            problems.append(f"message #{number:04d} has {len(reply_rows)} reply-required rows")
668        else:
669            reply_value = (
670                lines[reply_rows[0]].rstrip("\r\n").removeprefix("- 답필요: ")
671            )
672            if reply_value not in ("Y", "N"):
673                problems.append(
674                    f"message #{number:04d} has invalid reply-required value {reply_value!r}"
675                )
676
677        target_rows = metadata_field_lines(lines, start, end, "- 회신대상: ")
678        if len(target_rows) > 1:
679            problems.append(f"message #{number:04d} has {len(target_rows)} reply-target rows")
680        elif target_rows:
681            value = (
682                lines[target_rows[0]].rstrip("\r\n").removeprefix("- 회신대상: ")
683            )
684            try:
685                targets = parse_structured_reply_targets(value)
686            except ValueError as error:
687                problems.append(f"message #{number:04d} has invalid reply targets: {error}")
688            else:
689                prior_numbers = {prior for prior, _ in found[:index]}
690                for target in targets:
691                    if target not in prior_numbers:
692                        problems.append(
693                            f"message #{number:04d} reply target #{target:04d} "
694                            "must reference an existing earlier message"
695                        )
696    return not problems, problems, last

710@contextlib.contextmanager
711def mutation_lock(path: Path, timeout: float = LOCK_TIMEOUT_SECONDS) -> Iterator[None]:
712    """Serialize helper mutations with an exclusive sidecar lock."""
713    lock_path = path.with_name(f".{path.name}.mailbox.lock")
714    deadline = time.monotonic() + timeout
715    descriptor: int | None = None
716    while descriptor is None:
717        try:
718            descriptor = os.open(lock_path, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
719        except FileExistsError:
720            if time.monotonic() >= deadline:
721                raise TimeoutError(f"timed out waiting for mutation lock: {lock_path}")
722            time.sleep(0.05)
723    try:
724        os.write(descriptor, f"pid={os.getpid()} created={dt.datetime.now().isoformat()}\n".encode("ascii"))
725        os.fsync(descriptor)
726        yield
727    finally:
728        if descriptor is not None:
729            os.close(descriptor)
730        try:
731            lock_path.unlink()
732        except FileNotFoundError:
733            pass

736def atomic_write(path: Path, raw: bytes) -> None:
737    """Replace a file from the same directory after flushing the candidate."""
738    original_mode = stat.S_IMODE(path.stat().st_mode) if path.exists() else None
739    file_descriptor, temporary_name = tempfile.mkstemp(
740        prefix=f".{path.name}.", suffix=".tmp", dir=path.parent
741    )
742    temporary_path = Path(temporary_name)
743    try:
744        with os.fdopen(file_descriptor, "wb") as stream:
745            stream.write(raw)
746            stream.flush()
747            os.fsync(stream.fileno())
748        if original_mode is not None:
749            os.chmod(temporary_path, original_mode)
750        os.replace(temporary_path, path)
751    finally:
752        try:
753            temporary_path.unlink()
754        except FileNotFoundError:
755            pass

810def init_thread(path: Path, title: str, mode_value: str | None = None) -> int:
811    if path.exists():
812        print(f"[thread] refusing to overwrite existing file: {path}", file=sys.stderr)

816        print(f"[thread] template not found: {template}", file=sys.stderr)
817        return 2

819    raw = content.encode("utf-8")

832        return 3

834    atomic_write(path, raw)

884def resolve(path: Path, number: int, status_value: str, expected_mode: str | None) -> int:
885    with mutation_lock(path):
886        raw = read_bytes(path)
887        text = decode_utf8(raw)
888        ok, problems, _ = validate_raw(raw)
889        if not ok:
890            for problem in problems:
891                print(problem, file=sys.stderr)
892            return 2
893        mode_result = enforce_expected_mode(raw, expected_mode, "resolve")
894        if mode_result:
895            return mode_result
896        lines = text.splitlines(keepends=True)
897        found = parse_headers(text)
898        index = next((i for i, item in enumerate(found) if item[0] == number), None)
899        if index is None:
900            print(f"[thread] message not found: #{number:04d}", file=sys.stderr)
901            return 2
902        start = found[index][1] - 1
903        end = found[index + 1][1] - 1 if index + 1 < len(found) else len(lines)
904        state_lines = metadata_state_lines(lines, start, end)
905        if len(state_lines) != 1:
906            print(f"[thread] #{number:04d} has {len(state_lines)} status lines; refusing", file=sys.stderr)
907            return 2
908        old_line = lines[state_lines[0]]
909        newline = "\r\n" if old_line.endswith("\r\n") else "\n" if old_line.endswith("\n") else ""
910        lines[state_lines[0]] = f"- 상태: {status_value}{newline}"
911        candidate = "".join(lines)
912        candidate_raw = candidate.encode("utf-8")
913        ok, problems, _ = validate_raw(candidate_raw)
914        if not ok:
915            for problem in problems:
916                print(problem, file=sys.stderr)
917            return 3
918        atomic_write(path, candidate_raw)
919    print(with_awaiting(f"[thread] resolved #{number:04d} -> {status_value}", candidate))
920    return 0

933def append(path: Path, args: argparse.Namespace) -> int:
934    body = read_body(args)
935    if len(body) > MAX_INLINE_BODY_CHARS:
936        print(
937            f"[thread] append refused: body has {len(body)} chars; default maximum is "
938            f"{MAX_INLINE_BODY_CHARS}. Store the document in an artifact, then append a "
939            "short reply that links both --artifact and --in-reply-to.",
940            file=sys.stderr,
941        )
942        return 6
943    with mutation_lock(path):
944        existing_raw = read_bytes(path)
945        existing = decode_utf8(existing_raw)
946        ok, problems, last = validate_raw(existing_raw)
947        if not ok:
948            for problem in problems:
949                print(problem, file=sys.stderr)
950            return 2
951        mode_result = enforce_expected_mode(existing_raw, args.expect_mode, "append")
952        if mode_result:
953            return mode_result
954        if args.expect_last is not None and last != args.expect_last:
955            print(
956                f"[thread] append refused: expected last #{args.expect_last:04d}, "
957                f"actual last #{last:04d}; reread before retrying",
958                file=sys.stderr,
959            )
960            return 5
961
962        number = last + 1
963        try:
964            reply_targets = parse_cli_reply_targets(args.in_reply_to)
965        except ValueError as error:
966            print(f"[thread] append refused: {error}", file=sys.stderr)
967            return 2
968        artifacts = tuple(args.artifact or ())
969        if artifacts and not reply_targets:
970            print(
971                "[thread] append refused: --artifact requires --in-reply-to so the "
972                "artifact is connected to its target message",
973                file=sys.stderr,
974            )
975            return 2
976        for artifact in artifacts:
977            if not artifact.strip() or "\n" in artifact or "\r" in artifact:
978                print("[thread] append refused: artifact references must be one non-empty line", file=sys.stderr)
979                return 2
980        existing_numbers = {message_number for message_number, _ in parse_headers(existing)}
981        for target in reply_targets:
982            if target not in existing_numbers or target >= number:
983                print(
984                    f"[thread] append refused: reply target #{target:04d} must reference "
985                    "an existing earlier message",
986                    file=sys.stderr,
987                )
988                return 2
989
990        timestamp = args.date or dt.datetime.now().strftime("%Y-%m-%d %H:%M KST")
991        newline = newline_sequence(existing)
992        reply_target_line = ""
993        if reply_targets:
994            formatted_targets = ",".join(f"#{target:04d}" for target in reply_targets)
995            reply_target_line = f"- 회신대상: {formatted_targets}\n"
996        artifact_lines = "".join(
997            f"- 산출물: {artifact.strip()}\n" for artifact in artifacts
998        )
999        entry_lf = (
1000            f"### [{args.sender} → {args.receiver}] {timestamp} · {args.type} · "
1001            f"#{number:04d} · [{args.tag}] {args.title}\n"
1002            f"- 대상: {args.target}\n"
1003            f"- 답필요: {args.reply}\n"
1004            f"{reply_target_line}"
1005            f"{artifact_lines}"
1006            f"- 상태: {args.status}\n"
1007            "- 본문:\n"
1008            f"{body}\n"
1009        )
1010        entry = entry_lf if newline == "\n" else entry_lf.replace("\n", "\r\n")
1011        if not existing or existing.endswith(newline * 2):
1012            separator = ""
1013        elif existing.endswith(newline):
1014            separator = newline
1015        else:
1016            separator = newline * 2
1017        candidate = existing + separator + entry
1018        candidate_raw = candidate.encode("utf-8")
1019        ok, problems, candidate_last = validate_raw(candidate_raw, expected_last=number)
1020        if not ok or candidate_last != number:
1021            for problem in problems:
1022                print(problem, file=sys.stderr)
1023            return 3
1024        atomic_write(path, candidate_raw)
1025    print(with_awaiting(f"[thread] appended #{number:04d} -> {path}", candidate))
1026    return 0

1042def pin(path: Path, args: argparse.Namespace) -> int:
1043    if args.show_hash:
1044        raw = read_bytes(path)
1045        ok, problems, _ = validate_raw(raw)
1046        if not ok:
1047            for problem in problems:
1048                print(problem, file=sys.stderr)
1049            return 2
1050        mode_result = enforce_expected_mode(raw, args.expect_mode, "pin")
1051        if mode_result:
1052            return mode_result
1053        try:
1054            current_hash = pin_hash(raw)
1055        except ValueError as error:
1056            print(f"[thread] pin refused: {error}", file=sys.stderr)
1057            return 2
1058        print(with_awaiting(f"[thread] pin-hash {current_hash}", decode_utf8(raw)))
1059        return 0
1060
1061    if not args.expect_hash:
1062        print("[thread] pin update requires --expect-hash", file=sys.stderr)
1063        return 2
1064    body = read_body(args)
1065    with mutation_lock(path):
1066        raw = read_bytes(path)
1067        text = decode_utf8(raw)
1068        ok, problems, _ = validate_raw(raw)
1069        if not ok:
1070            for problem in problems:
1071                print(problem, file=sys.stderr)
1072            return 2
1073        mode_result = enforce_expected_mode(raw, args.expect_mode, "pin")
1074        if mode_result:
1075            return mode_result
1076        try:
1077            layout_before = thread_layout(raw)
1078            if layout_before.pin is None:
1079                raise ValueError("pin requires markers in the pre-message region")
1080            body_start = layout_before.pin.body_start
1081            body_end = layout_before.pin.body_end
1082            old_body = layout_before.pin.body
1083        except ValueError as error:
1084            print(f"[thread] pin refused: {error}", file=sys.stderr)
1085            return 2
1086        current_hash = hashlib.sha256(old_body).hexdigest()
1087        if current_hash != args.expect_hash:
1088            print(
1089                f"[thread] pin refused: expected hash {args.expect_hash}, "
1090                f"actual hash {current_hash}",
1091                file=sys.stderr,
1092            )
1093            return 5
1094
1095        newline = newline_sequence(text)
1096        normalized_body = body.replace("\r\n", "\n").replace("\r", "\n").strip("\n")
1097        if newline == "\r\n":
1098            normalized_body = normalized_body.replace("\n", "\r\n")
1099        replacement = (normalized_body + newline).encode("utf-8")
1100        candidate_raw = raw[:body_start] + replacement + raw[body_end:]
1101        try:
1102            layout_after = thread_layout(candidate_raw)
1103        except ValueError as error:
1104            print(f"[thread] pin refused: {error}", file=sys.stderr)
1105            return 3
1106        mode_before = marker_block_bytes(raw, layout_before.mode)
1107        mode_after = marker_block_bytes(candidate_raw, layout_after.mode)
1108        messages_before = raw[layout_before.first_message :]
1109        messages_after = candidate_raw[layout_after.first_message :]
1110        if mode_after != mode_before:
1111            print("[thread] pin refused: mode bytes would change", file=sys.stderr)
1112            return 3
1113        if messages_after != messages_before:
1114            print("[thread] pin refused: message bytes would change", file=sys.stderr)
1115            return 3
1116        ok, problems, _ = validate_raw(candidate_raw)
1117        if not ok:
1118            for problem in problems:
1119                print(problem, file=sys.stderr)
1120            return 3
1121        atomic_write(path, candidate_raw)
1122        updated_hash = pin_hash(candidate_raw)
1123    print(
1124        with_awaiting(
1125            f"[thread] pin updated hash={updated_hash}", decode_utf8(candidate_raw)
1126        )
1127    )
1128    return 0

1212                print(
1213                    f"[thread] mode set refused: expected current mode "
1214                    f"{args.expect_current_mode}, actual {current.value}",
1215                    file=sys.stderr,
1216                )
1217                return 5
1218            if args.expect_hash is not None and args.expect_hash != current.body_hash:
1219                print(
1220                    f"[thread] mode set refused: expected hash {args.expect_hash}, "
1221                    f"actual {current.body_hash}",
1222                    file=sys.stderr,
1223                )
1224                return 5
1225
1226        text = decode_utf8(raw)
1227        newline = newline_sequence(text)
1228        replacement_body = formatted_mode_body(args.set_mode, setter, reason, newline)
1229        if layout_before.mode is None:
1230            assert layout_before.pin is not None
1231            block = formatted_mode_block(args.set_mode, setter, reason, newline)
1232            insert_at = layout_before.pin.start
1233            candidate_raw = raw[:insert_at] + block + raw[insert_at:]
1234        else:
1235            candidate_raw = (
1236                raw[: layout_before.mode.body_start]
1237                + replacement_body
1238                + raw[layout_before.mode.body_end :]
1239            )
1240
1241        try:
1242            layout_after = thread_layout(candidate_raw)
1243            updated = mode_info(candidate_raw)
1244        except ValueError as error:
1245            print(f"[thread] mode refused: {error}", file=sys.stderr)
1246            return 3
1247        if updated is None or updated.value != args.set_mode:
1248            print("[thread] mode refused: candidate mode validation failed", file=sys.stderr)
1249            return 3
1250        pin_before = marker_block_bytes(raw, layout_before.pin)
1251        pin_after = marker_block_bytes(candidate_raw, layout_after.pin)
1252        messages_before = raw[layout_before.first_message :]
1253        messages_after = candidate_raw[layout_after.first_message :]
1254        if pin_after != pin_before:
1255            print("[thread] mode refused: pinned bytes would change", file=sys.stderr)
1256            return 3
1257        if messages_after != messages_before:
1258            print("[thread] mode refused: message bytes would change", file=sys.stderr)
1259            return 3
1260        ok, problems, _ = validate_raw(candidate_raw)
1261        if not ok:
1262            for problem in problems:
1263                print(problem, file=sys.stderr)
1264            return 3
1265        atomic_write(path, candidate_raw)
1266
1267    print(
1268        with_awaiting(
1269            f"[thread] mode updated mode={updated.value} hash={updated.body_hash}",
1270            decode_utf8(candidate_raw),
1271        )
1272    )
1273    return 0

1396def main() -> int:
1397    try:
1398        sys.stdout.reconfigure(encoding="utf-8", line_buffering=True)
1399        sys.stderr.reconfigure(encoding="utf-8", line_buffering=True)
1400    except Exception:
1401        pass
1402    args = build_parser().parse_args()
1403    path = args.thread.resolve()
1404    if args.command == "init":
1405        return init_thread(path, args.title, args.mode)
1406    if args.command == "validate":
1407        ok, problems, last = validate(path)
1408        text = read_text(path) if path.exists() else ""
1409        if ok:
1410            count = len(parse_headers(text))
1411            raw = read_bytes(path)
1412            info = mode_info(raw)
1413            mode_label = info.value if info is not None else "UNDECLARED"
1414            print(
1415                with_awaiting(
1416                    f"[thread] ok messages={count} last=#{last:04d} "
1417                    f"{newline_summary(text)} mode={mode_label}",
1418                    text,
1419                )
1420            )
1421            return 0
1422        for problem in problems:
1423            print(problem, file=sys.stderr)
1424        if path.exists():
1425            print(f"[thread] {newline_summary(text)}", file=sys.stderr)
1426        return 1
1427    if args.command == "open":
1428        text = read_text(path)
1429        if args.awaiting:
1430            awaiting = awaiting_reply_messages(text)
1431            if args.after_number is not None and args.after_number < 0:
1432                print(
1433                    "[thread] open refused: --after-number must be 0 or greater",
1434                    file=sys.stderr,
1435                )
1436                return 2
1437            awaiting = [
1438                message
1439                for message in awaiting
1440                if receiver_matches(message.receiver, args.agent)
1441                and (
1442                    args.after_number is None
1443                    or message.number > args.after_number
1444                )
1445            ]
1446            if args.agent is not None or args.after_number is not None:
1447                after_label = (
1448                    f"#{args.after_number:04d}"
1449                    if args.after_number is not None
1450                    else "NONE"
1451                )
1452                print(
1453                    f"[thread] awaiting agent={args.agent or 'ALL'} "
1454                    f"after={after_label} matched={len(awaiting)}"
1455                )
1456            if not awaiting:
1457                print(with_awaiting("[thread] no awaiting reply messages", text))
1458            for message in awaiting:
1459                print(
1460                    with_awaiting(
1461                        f"#{message.number:04d} · {message.sender} → "
1462                        f"{message.receiver} · {message.title}",
1463                        text,
1464                    )
1465                )
1466            return 0
1467        if args.agent is not None or args.after_number is not None:
1468            print(
1469                "[thread] open refused: --agent and --after-number require --awaiting",
1470                file=sys.stderr,
1471            )
1472            return 2
1473        statuses = PENDING_STATUSES if args.all else ("OPEN",)
1474        pending = messages_with_status(path, statuses)
1475        if not pending:
1476            label = "pending" if args.all else "OPEN"
1477            print(with_awaiting(f"[thread] no {label} messages", text))
1478        for number, status_value, header in pending:
1479            print(with_awaiting(f"#{number:04d} [{status_value}] {header}", text))
1480        return 0
1481    if args.command == "intake":
1482        return intake(path, args)
1483    if args.command == "show":
1484        return show_message(path, args.number, args.with_links, args.depth)
1485    if args.command == "next":
1486        ok, problems, last = validate(path)
1487        structural_problems = [
1488            problem for problem in problems if not problem.startswith("mixed line endings:")
1489        ]
1490        if structural_problems:
1491            for problem in structural_problems:
1492                print(problem, file=sys.stderr)
1493            return 1
1494        if not ok:
1495            print(
1496                "[thread] warning: mixed newlines detected; read-only next is allowed, "
1497                "mutation commands will refuse",
1498                file=sys.stderr,
1499            )
1500        print(f"{last + 1:04d}")
1501        return 0
1502    if args.command == "resolve":
1503        return resolve(path, args.number, args.status, args.expect_mode)
1504    if args.command == "pin":
1505        return pin(path, args)
1506    if args.command == "mode":
1507        return mode_command(path, args)
1508    return append(path, args)
